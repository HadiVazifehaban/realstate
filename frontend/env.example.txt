# =================================================================
# Maskanchi API (FastAPI office node)
# =================================================================
# Origin of the FastAPI office node, as reachable from the Next.js server.
# Server-side only (never NEXT_PUBLIC_): the browser reaches the node through a
# same-origin /backend/* rewrite defined in next.config.ts, so nothing about the
# node's address is baked into the client bundle. That is what keeps the app
# working on a phone over the LAN, where a baked-in localhost would resolve to
# the phone itself.
#
# Inside compose the backend is reachable by service name. Running the frontend
# outside compose, point this at the published port instead:
#   API_SERVER_ORIGIN=http://localhost:8080

API_SERVER_ORIGIN=http://backend:8080


# =================================================================
# Map tiles (self-hosted OpenStreetMap)
# =================================================================
# The `tiles` service in ../docker-compose.yml runs
# overv/openstreetmap-tile-server on port 8081 and serves
# {base}/{z}/{x}/{y}.png — note the path segment is singular: /tile, not /tiles.
# The default below matches that local install.
#
# This is a public value on purpose: raster tiles carry no owner or client PII.
# On a LAN office install point it at the server's LAN address, otherwise
# `localhost` resolves to the *viewer's* machine and the map comes up blank.

NEXT_PUBLIC_TILE_URL=http://localhost:8081/tile


# =================================================================
# Keycloak / OIDC identity (MASKANCHI-SEC-KEYCLOAK-2026-V1)
# =================================================================
# Setting NEXT_PUBLIC_KEYCLOAK_ISSUER switches the dashboard to the Keycloak
# stack. Leave it EMPTY to keep the Clerk (or bypass) mode.
#
# It must match the API core's KEYCLOAK_ISSUER exactly -- the core pins token
# verification to that issuer and to KEYCLOAK_AUDIENCE, so a mismatch makes
# every request fail signature verification.
#
# NOTE: /dashboard/users reads the Keycloak realm whenever the API has a
# service account configured (KEYCLOAK_ADMIN_CLIENT_ID/SECRET in the API's
# .env). This browser-side issuer only controls how the dashboard *logs in*;
# the directory itself is fetched through the API, never from the browser.
NEXT_PUBLIC_KEYCLOAK_ISSUER=
KEYCLOAK_CLIENT_ID=maskanchi-frontend

# Public client with PKCE, so no secret is required. A public client cannot keep
# a secret: anything in a browser bundle is not a secret.
KEYCLOAK_CLIENT_SECRET=

# Must match the API core's KEYCLOAK_AUDIENCE. A token minted for another
# audience is rejected with token_audience_mismatch.
KEYCLOAK_AUDIENCE=maskanchi-backend-api

# Self-hosted Auth.js session signing key (openssl rand -base64 32).
NEXTAUTH_SECRET=
# Do not set NEXTAUTH_URL: Auth.js derives callback URLs from the validated
# incoming host, allowing localhost, LAN IPs and VPS origins on one deployment.
# Set AUTH_CANONICAL_URL only if Keycloak logout should always return to one
# registered URL (for example, the public VPS HTTPS domain).
# AUTH_CANONICAL_URL=https://crm.example.com
AUTH_TRUST_HOST=true



# =================================================================
# Development-only auth bypass (READ THIS BEFORE DEPLOYING)
# =================================================================
# The dashboard starter requires Clerk keys. With the keys absent,
# ClerkProvider/clerkMiddleware throw at render time and the whole app
# returns HTTP 500 — which blocks local UI work unrelated to auth.
#
# Setting this to "true" skips Clerk locally so screens such as the
# Maskanchi radar (which only talks to the LAN FastAPI node) can be opened.
#
# ⚠ SECURITY: this disables authentication entirely. It is client-visible and
# meant ONLY for local development. Set it to "false"/remove it before any
# shared or deployed environment, and add real Clerk keys there.
#
# NOTE: default is "false" — this project runs Keycloak as the real gate.
# `true` makes "Sign out" a no-op that bounces back to the dashboard, and it is
# inlined into the client bundle at BUILD time, so flipping it requires a
# frontend image rebuild.
NEXT_PUBLIC_AUTH_BYPASS=false
